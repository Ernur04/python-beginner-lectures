Декораторы

def timer(func):
        import time
        def wrapper(*args, **kwargs):
            start = time.time()
            result = func(*args, **kwargs)
            print(f"Время: {time.time()-start:.2f}")
            return result
        return wrapper

    @timer
    def slow_function():
        time.sleep(1)
        return "Готово"

    slow_function()

Модификация функций без изменения кода.

Генераторы

# Генераторная функция
    def countdown(n):
        while n > 0:
            yield n
            n -= 1

    # Использование
    for num in countdown(3):
        print(num)  # 3, 2, 1

    # Generator expression
    squares = (x**2 for x in range(5))
